llvm::Value* LLVM辅助::读取行并剥离换行(llvm::Value* 文件指针,
                                       llvm::IRBuilder<>& 构建器, llvm::Module& 模块, llvm::LLVMContext& 上下文) {
    auto* fgets = C运行时声明::获取fgets(模块, 上下文);
    auto* malloc = C运行时声明::获取malloc(模块, 上下文);
    auto* strlen = C运行时声明::获取strlen(模块, 上下文);
    
    llvm::Value* 缓冲区 = 构建器.CreateCall(malloc,
        {llvm::ConstantInt::get(上下文, llvm::APInt(64, 4096))}, "读取行缓冲区");
    构建器.CreateCall(fgets, {缓冲区,
        llvm::ConstantInt::get(上下文, llvm::APInt(32, 4096)), 文件指针});

        llvm::ConstantInt::get(上下文, llvm::APInt(64, 0)));
    llvm::Value* 最后位置 = 构建器.CreateSub(长度,
        llvm::ConstantInt::get(上下文, llvm::APInt(64, 1)));
    llvm::Value* 安全位置 = 构建器.CreateSelect(长度大于零, 最后位置,
        llvm::ConstantInt::get(上下文, llvm::APInt(64, 0)));
    llvm::Value* 最后字符地址 = 构建器.CreateInBoundsGEP(
        llvm::Type::getInt8Ty(上下文), 缓冲区, 安全位置);
    llvm::Value* 最后字符 = 构建器.CreateLoad(
        llvm::Type::getInt8Ty(上下文), 最后字符地址);
    llvm::Value* 是换行 = 构建器.CreateICmpEQ(最后字符,
        llvm::ConstantInt::get(上下文, llvm::APInt(8, '\n')));
    llvm::Value* 应替换 = 构建器.CreateAnd(长度大于零, 是换行);
    llvm::Value* 空终止 = llvm::ConstantInt::get(llvm::Type::getInt8Ty(上下文), 0);
    llvm::Value* 原始字节 = 构建器.CreateLoad(
        llvm::Type::getInt8Ty(上下文), 最后字符地址);
    llvm::Value* 最终字节 = 构建器.CreateSelect(应替换, 空终止, 原始字节);

### 2.4 提取 `查找结构体成员地址` （消除 3 处重复）

**位置**: `代码生成器_语句.cpp`

**重复处**:
1. 赋值语句中的成员赋值 (行 104-121)
2. 成员访问表达式 (代码生成器_表达式.cpp 中的 `成员访问` case)
3. (隐含在其他结构体操作中)

**提取为**:
```cpp
llvm::Value* LLVM辅助::查找结构体成员地址(const std::string& 对象名, const std::string& 成员名,
                                         符号表& 符号表实例,
                                         std::unordered_map<std::string, llvm::StructType*>& 结构体类型映射,
                                         std::unordered_map<std::string, std::vector<std::string>>& 结构体成员映射,
                                         llvm::IRBuilder<>& 构建器, llvm::LLVMContext& 上下文) {
    llvm::Value* 对象地址 = 符号表实例.获取变量值(对象名);
    if (!对象地址) throw std::runtime_error("未定义的变量: " + 对象名);
    const std::string& 结构体名 = 符号表实例.获取结构体类型(对象名);
    if (结构体名.empty()) throw std::runtime_error("变量不是结构体: " + 对象名);
    auto it = 结构体类型映射.find(结构体名);
    if (it == 结构体类型映射.end()) throw std::runtime_error("未定义的结构体: " + 结构体名);
    llvm::StructType* 结构体类型 = it->second;
    const auto& 成员名列表 = 结构体成员映射[结构体名];
    int 成员索引 = -1;
    for (size_t i = 0; i < 成员名列表.size(); i++) {
        if (成员名列表[i] == 成员名) { 成员索引 = static_cast<int>(i); break; }
    }
    if (成员索引 < 0) throw std::runtime_error("未定义的成员: " + 成员名);
    return 构建器.CreateStructGEP(结构体类型, 对象地址, static_cast<unsigned>(成员索引), 成员名);
}
```

llvm::Type* LLVM辅助::计算返回类型(const std::vector<返回值描述>& 返回值列表,

                                   std::function<llvm::Type*(const std::string&)> 类型转换器) {
    llvm::Type* 返回LLVM类型 = llvm::Type::getVoidTy(上下文);
    if (返回值列表.size() == 1 && !返回值列表[0].类型.empty() && 返回值列表[0].类型 != "空") {

        }

### 2.6 提取 `短路逻辑运算` （消除 ~40 行重复）

**位置**: `代码生成器_表达式.cpp` 行 60-103

**当前**: `逻辑或` 和 `逻辑与` 各 ~20 行，结构几乎相同，仅 PHI 默认值和分支方向不同。

**提取为**:
```cpp
llvm::Value* LLVM辅助::短路逻辑运算(二元操作符 操作符,
                                  llvm::Value* 左, std::function<llvm::Value*()> 生成右,
                                  llvm::IRBuilder<>& 构建器, llvm::LLVMContext& 上下文) {
    if (左->getType()->isIntegerTy(1)) 左 = 构建器.CreateZExt(左, llvm::Type::getInt32Ty(上下文));
    llvm::Value* 左条件 = 构建器.CreateICmpNE(左,
        llvm::ConstantInt::get(上下文, llvm::APInt(32, 0)), "sc_cond");
    llvm::Function* 当前函数 = 构建器.GetInsertBlock()->getParent();
    llvm::BasicBlock* 右块 = llvm::BasicBlock::Create(上下文, "sc_rhs", 当前函数);
    llvm::BasicBlock* 合并块 = llvm::BasicBlock::Create(上下文, "sc_merge");
    
    bool 是与 = (操作符 == 二元操作符::逻辑与);
    // 与: 左真→右块, 左假→合并(假)
    // 或: 左真→合并(真), 左假→右块
    llvm::BasicBlock* 真目标 = 是与 ? 右块 : 合并块;
    llvm::BasicBlock* 假目标 = 是与 ? 合并块 : 右块;
    构建器.CreateCondBr(左条件, 真目标, 假目标);
    
    llvm::BasicBlock* 左块 = 构建器.GetInsertBlock();
    构建器.SetInsertPoint(右块);
    llvm::Value* 右 = 生成右();
    if (右->getType()->isIntegerTy(1)) 右 = 构建器.CreateZExt(右, llvm::Type::getInt32Ty(上下文));
    llvm::Value* 右条件 = 构建器.CreateICmpNE(右,
        llvm::ConstantInt::get(上下文, llvm::APInt(32, 0)), "sc_cond2");
    右块 = 构建器.GetInsertBlock();
    构建器.CreateBr(合并块);
    当前函数->insert(当前函数->end(), 合并块);
    构建器.SetInsertPoint(合并块);
    
    llvm::PHINode* phi = 构建器.CreatePHI(llvm::Type::getInt1Ty(上下文), 2, "sc");
    phi->addIncoming(llvm::ConstantInt::get(上下文, 是与 ? llvm::APInt(1, 0) : llvm::APInt(1, 1)), 左块);
    phi->addIncoming(右条件, 右块);
    return 构建器.CreateZExt(phi, llvm::Type::getInt32Ty(上下文), "sc_ext");
}
```

- `正弦/余弦/正切` → 内部调用 `sin/cos/tan`

当前实现已经正确（C 包装函数存在），只需消除 LLVM codegen 中的重复声明。

### 4.1 实现 do-while 循环 (`做循环语句`)

**当前状态**: 
- 解析器正确解析 `做 { ... } 当(条件);` 和 `做 { ... } 到(条件);`
- LLVM codegen 抛出异常："做循环暂不支持编译模式"

**实现方案** (`代码生成器_语句.cpp` 行 282-286):

```cpp
case 语句类型::做循环语句: {
    const auto& 做循环 = static_cast<const 做循环语句&>(语句);
    llvm::Function* 当前函数 = 构建器->GetInsertBlock()->getParent();
    
    llvm::BasicBlock* 体块 = llvm::BasicBlock::Create(上下文, "dowhile_body", 当前函数);
    llvm::BasicBlock* 条件块 = llvm::BasicBlock::Create(上下文, "dowhile_cond", 当前函数);
    llvm::BasicBlock* 后块 = llvm::BasicBlock::Create(上下文, "dowhile_end", 当前函数);
    
    循环继续栈.push(条件块);
    循环退出栈.push(后块);
    
    // 先执行循环体
    构建器->CreateBr(体块);
    构建器->SetInsertPoint(体块);
    符号表实例.进入作用域();
    for (const auto& 子 : 做循环.主体) 生成语句(*子);
    符号表实例.退出作用域();
    if (构建器->GetInsertBlock()->getTerminator() == nullptr)
        构建器->CreateBr(条件块);
    
    // 再检查条件
    构建器->SetInsertPoint(条件块);
    llvm::Value* 条件值 = 生成表达式(*做循环.条件);
    条件值 = 构建器->CreateICmpNE(条件值,
        llvm::ConstantInt::get(上下文, llvm::APInt(32, 0)), "docond");
    
    if (做循环.是否当循环) {
        // 做-当: 条件为真时继续
        构建器->CreateCondBr(条件值, 体块, 后块);
    } else {
        // 做-到: 条件为真时退出（"到" = until）
        构建器->CreateCondBr(条件值, 后块, 体块);
    }
    
    循环继续栈.pop();
    循环退出栈.pop();
    构建器->SetInsertPoint(后块);
    break;
}
```

**关键点**:
- 与 while 的区别：先执行体块，再跳到条件块
- `做-当` = do-while（条件真继续），`做-到` = do-until（条件真退出）
- 体块中的 `继续` 跳到条件块，`中断` 跳到后块

**测试**: 编写 `示例/做循环测试.lune` 验证编译模式。

### 4.2 实现 `return` 语句

**当前状态**: 
- 解析器正确解析 `返回 值;` 和 `返回;`
- LLVM codegen 抛出异常："不允许使用'返回'关键字，请直接赋值返回值变量"

**实现方案** (`代码生成器_语句.cpp` 行 301-303):

需要在 `代码生成器` 类中添加成员变量跟踪当前函数的返回值分配列表：

```cpp
// 代码生成器_内部.h 中添加
std::vector<llvm::AllocaInst*> 当前返回值分配列表;
```

然后在 `生成函数` 中设置它，在 `返回语句` case 中使用：

```cpp
case 语句类型::返回语句: {
    const auto& 返回 = static_cast<const 返回语句&>(语句);
    if (构建器->GetInsertBlock()->getTerminator() != nullptr) break;
    
    if (返回.返回值) {
        llvm::Value* 值 = 生成表达式(*返回.返回值);
        if (当前返回值分配列表.size() == 1) {
            构建器->CreateStore(值, 当前返回值分配列表[0]);
        }
        // 多返回值情况：如果返回的是结构体，拆解存储
        // 否则存储到第一个返回值
    }
    
    // 跳转到函数末尾（需要统一出口块）
    if (!当前函数出口块) {
        当前函数出口块 = llvm::BasicBlock::Create(上下文, "return_exit",
            构建器->GetInsertBlock()->getParent());
    }
    构建器->CreateBr(当前函数出口块);
    
    // 创建新块继续生成后续代码（死代码，但避免 crash）
    llvm::BasicBlock* 新块 = llvm::BasicBlock::Create(上下文, "after_return",
        构建器->GetInsertBlock()->getParent());
    构建器->SetInsertPoint(新块);
    break;
}
```

还需要修改 `生成函数` 末尾的自动返回逻辑，改为在出口块中生成：

```cpp
// 在 生成函数 末尾
if (当前函数出口块) {
    // 如果当前块没有终结指令，跳转到出口块
    if (构建器->GetInsertBlock()->getTerminator() == nullptr)
        构建器->CreateBr(当前函数出口块);
    
    构建器->SetInsertPoint(当前函数出口块);
    // 在出口块中生成返回指令
    if (返回值分配列表.size() == 1) {
        llvm::Value* 返回值 = 构建器->CreateLoad(...);
        构建器->CreateRet(返回值);
    } else if (返回值分配列表.size() > 1) {
        // 多返回值结构体
        ...
        构建器->CreateRet(返回结构体);
    } else {
        构建器->CreateRetVoid();
    }
}
```

**设计决策**: 使用统一出口块（single exit block）模式，所有 `返回` 语句跳转到出口块，
出口块中统一生成 `ret` 指令。这符合 LLVM 最佳实践，也便于后续添加清理代码。

### 4.3 实现 `import` 语句

**当前状态**: 
- 解析器正确解析 `导入 "路径";`、`导入 "路径" as 别名;`、`导入 "路径" { 函数1, 函数2 };`
- LLVM codegen 是空操作（`case 语句类型::导入语句: break;`）

**实现方案**: 分两步

#### 步骤 1: 在解析阶段处理导入

在 `语法分析器::解析程序()` 中，当遇到 `导入 "xxx";` 时：
1. 读取目标文件
2. 创建新的词法分析器和语法分析器
3. 解析目标文件为 AST（`程序`）
4. 将目标程序的函数列表、结构体定义、枚举定义合并到当前程序

```cpp
// 在 解析程序() 的导入处理中（已有框架，行 1540-1579）
} else if (_当前标记.类型 == 标记类型::导入) {
    前进();
    期望类型(标记类型::字符串, "导入后应为文件路径字符串");
    std::string 模块路径 = _当前标记.值;
    前进();
    // ... 现有的别名和选择性导入解析 ...
    期望(标记类型::分号, "导入后应有';'");
    
    // 读取并解析导入的文件
    std::string 完整路径 = 查找模块文件(模块路径);
    std::string 内容 = 读取文件内容(完整路径);
    语法分析器 子分析器(内容.c_str(), 内容.size());
    auto 子程序 = 子分析器.解析程序();
    
    // 合并到当前程序
    for (auto& 函数 : 子程序->函数列表) {
        程序->函数列表.push_back(std::move(函数));
    }
    for (auto& [名, 成员] : 子程序->结构体定义列表) {
        程序->结构体定义列表.push_back({名, std::move(成员)});
    }
    for (auto& [名, 成员] : 子程序->枚举定义列表) {
        程序->枚举定义列表.push_back({名, std::move(成员)});
    }
}
```

#### 步骤 2: 添加文件读取辅助

在 `语法分析器.h` 中添加：
```cpp
static std::string 查找模块文件(const std::string& 模块路径);
static std::string 读取文件内容(const std::string& 文件路径);
```

实现：
```cpp
std::string 语法分析器::查找模块文件(const std::string& 模块路径) {
    // 1. 相对路径（相对于当前文件）
    // 2. 标准库路径（标准库/）
    // 3. 绝对路径
    // 优先尝试 .lune 扩展名
    std::vector<std::string> 候选 = {
        模块路径,
        模块路径 + ".lune",
        "标准库/" + 模块路径,
        "标准库/" + 模块路径 + ".lune",
    };
    for (const auto& 路径 : 候选) {
        if (文件存在(路径)) return 路径;
    }
    throw std::runtime_error("找不到模块: " + 模块路径);
}
```

**注意**: import 实现在解析阶段（前端），而非 codegen 阶段。这样 LLVM codegen 和字节码编译器
都能自动获得导入的函数，无需各自实现。codegen 中的 `导入语句` case 保持为空操作即可。
